28// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept
29// the exact signature, `noexcept` and the result type; the driver rejects a
30// hook that is declared but does not match (handbook KB-CPP20-007).
31
34template <typename Port>
35concept DeclaresGpioSet =
36 requires(Port& port) { port.GpioSet(CtrlPin::Enable, GpioSignal::ACTIVE); };
37
39template <typename Port>
40concept HasGpioSet =
41 requires { ExactHook<Port, bool(CtrlPin, GpioSignal) noexcept>::Take(&Port::GpioSet); };
42
44template <typename Port>
45concept DeclaresDelayUs = requires(Port& port) { port.DelayUs(1_u32); };
46
48template <typename Port>
49concept HasDelayUs = requires { ExactHook<Port, void(uint32_t) noexcept>::Take(&Port::DelayUs); };
50
52template <typename Port>
53concept DeclaresLog = requires(Port& port) { port.Log(LogLevel::Info, ""); };
54
57template <typename Port>
58concept HasLog =
59 requires { ExactHook<Port, void(LogLevel, const char*) noexcept>::Take(&Port::Log); };

63template <typename Port>
64void DelayUs(Port& port, uint32_t us) noexcept {
65 if constexpr (HasDelayUs<Port>) {
66 port.DelayUs(us);
67 } else {
68 static_assert(kAlwaysFalse<Port>, "mcp9700: the port must implement `void "
69 "DelayUs(uint32_t us) noexcept`");
70 }
