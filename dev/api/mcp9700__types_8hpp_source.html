468constexpr int64_t RoundDivAway(int64_t num, int64_t den) noexcept {
469 const int64_t half = den / 2;
470 return (num >= 0) ? ((num + half) / den) : (-((-num + half) / den));
471}

474constexpr int64_t OutputNanoVolts(uint32_t v0_mv, uint32_t tc_uv_per_c, int64_t t_mc) noexcept {
475 return (static_cast<int64_t>(v0_mv) * kPerMega) + (static_cast<int64_t>(tc_uv_per_c) * t_mc);

484constexpr bool CustomIsZero(const CustomSpec& c) noexcept {
485 return (c.v0_mv == 0U) && (c.tc_uv_per_c == 0U) && (c.min_mc == 0) && (c.max_mc == 0) &&

491constexpr bool CustomOutputInWindow(const CustomSpec& c, int32_t t_mc) noexcept {
492 const int64_t nv = OutputNanoVolts(c.v0_mv, c.tc_uv_per_c, t_mc);
493 return (nv >= 0) && (nv <= (static_cast<int64_t>(kCustomMaxOutputMv) * kPerMega));

497constexpr bool CustomTransferOk(const CustomSpec& c) noexcept {
498 return (c.tc_uv_per_c >= kCustomMinTcUvPerC) && (c.tc_uv_per_c <= kCustomMaxTcUvPerC) &&
499 (c.v0_mv <= kCustomMaxV0Mv) && (c.min_mc < c.max_mc) &&
500 WithinSymmetric(c.min_mc, kCustomMaxTemperatureMc) &&
501 WithinSymmetric(c.max_mc, kCustomMaxTemperatureMc) && CustomOutputInWindow(c, c.min_mc) &&
502 CustomOutputInWindow(c, c.max_mc);
503}
504
507constexpr bool CustomAccuracyOk(const CustomSpec& c) noexcept {
508 return (c.accuracy_min_mc <= 0) && (c.accuracy_max_mc >= 0) &&
509 ((c.accuracy_min_mc != 0) || (c.accuracy_max_mc != 0)) &&
510 WithinSymmetric(c.accuracy_min_mc, kCustomMaxAccuracyMc) &&
511 WithinSymmetric(c.accuracy_max_mc, kCustomMaxAccuracyMc);
512}

516constexpr bool VariantDeclarationOk(const Config& config) noexcept {
517 if (!IsEnumerator(config.variant) || !IsEnumerator(config.range)) {
518 return false;
519 }
520 if (config.variant == Variant::Custom) {
521 return (config.range == TemperatureClass::Extended) && CustomTransferOk(config.custom) &&
522 CustomAccuracyOk(config.custom);
523 }
524 if (!CustomIsZero(config.custom)) {
525 return false;
526 }
527 return (config.range != TemperatureClass::High) ||
528 kVariantTable[static_cast<std::size_t>(ToUnderlying(config.variant))].has_high;
529}
530
532constexpr bool AdcDeclarationOk(const Config& config) noexcept {
533 return (config.adc_full_scale_mv >= kMinAdcFullScaleMv) &&
534 (config.adc_full_scale_mv <= kMaxAdcFullScaleMv) &&
535 (config.adc_resolution_bits >= kMinAdcResolutionBits) &&
536 (config.adc_resolution_bits <= kMaxAdcResolutionBits) && (config.divider_num != 0U) &&
537 (config.divider_den != 0U) && WithinSymmetric(config.offset_cal_mc, kMaxOffsetCalMc) &&
538 (config.range_margin_mc >= 0) && (config.range_margin_mc <= kMaxRangeMarginMc);
539}
540
542constexpr bool SamplingOk(const Config& config) noexcept {
543 if ((config.samples == 0U) || (config.samples > kMaxSamples) ||
544 (config.discard_conversions > kMaxDiscard) ||
545 (config.sample_interval_us > kMaxSampleIntervalUs)) {
546 return false;
547 }
548 const uint64_t conversions = static_cast<uint64_t>(config.discard_conversions) +
549 static_cast<uint64_t>(config.samples) - 1U;
550 return ((conversions * static_cast<uint64_t>(config.sample_interval_us)) <=
551 kMaxAveragingWindowUs) &&
552 (config.max_spread_mv <= kMaxSpreadLimitMv) &&
553 (config.power_on_settle_us <= kMaxPowerOnSettleUs);
554}

580constexpr Result<VariantSpec> ResolveSpec(const Config& config) noexcept {
581 if (Validate(config) != Error::Ok) {

608constexpr int32_t OutputMvToMilliC(const VariantSpec& spec, uint32_t output_mv) noexcept {
609 constexpr uint32_t kLargest = 65'535U;
610 const uint32_t mv = (output_mv < kLargest) ? output_mv : kLargest;
611 const int64_t num =
612 (static_cast<int64_t>(mv) - static_cast<int64_t>(spec.v0_mv)) * detail::kPerMega;
613 return static_cast<int32_t>(detail::RoundDivAway(num, static_cast<int64_t>(spec.tc_uv_per_c)));
614}

623constexpr uint32_t MilliCToOutputMv(const VariantSpec& spec, int32_t temperature_mc) noexcept {
624 const int64_t nv = detail::OutputNanoVolts(spec.v0_mv, spec.tc_uv_per_c, temperature_mc);
625 if (nv <= 0) {
626 return 0U;
627 }
628 const int64_t mv = detail::RoundDivAway(nv, detail::kPerMega);
629 constexpr int64_t kLargest = 0xFFFF'FFFFLL;
630 return static_cast<uint32_t>((mv < kLargest) ? mv : kLargest);
631}

645constexpr Result<AccuracyLimits> AccuracyLimitsFor(const VariantSpec& spec, TemperatureClass range,
646 int32_t temperature_mc) noexcept {
647 const bool high = range == TemperatureClass::High;
648 const int32_t top = high ? spec.max_high_mc : spec.max_ext_mc;
649 if ((high && !spec.has_high) || (temperature_mc < spec.min_mc) || (temperature_mc > top)) {
650 return Result<AccuracyLimits>::Failure(Error::InvalidArgument);
651 }
652 for (std::size_t i = 0; i < kMaxBands; ++i) {
653 const AccuracyBand& band = spec.bands[i];
654 if ((i < static_cast<std::size_t>(spec.band_count)) && (band.high_only == high) &&
655 (temperature_mc >= band.from_mc) && (temperature_mc <= band.to_mc)) {
656 return Result<AccuracyLimits>::Success({.min_mc = band.min_mc, .max_mc = band.max_mc});
657 }
658 }
659 return Result<AccuracyLimits>::Failure(Error::InvalidArgument);
660}
